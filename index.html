<script>

  let db;

  const dbName = "OdysseusDB";

  let currentDate = new Date().toISOString().split('T')[0];

  let currentWeekStart = getMonday(new Date());

  let currentMonthDate = new Date();

  let isUnlocked = false;

  let currentView = 'daily';

  let activeTaskIdForAI = null;

  let isRunningRecoveryMode = false;


  function initDB() {

    const request = indexedDB.open(dbName, 1);

    request.onupgradeneeded = (e) => {

      db = e.target.result;

      if (!db.objectStoreNames.contains('tasks')) db.createObjectStore('tasks', { keyPath: 'id', autoIncrement: true });

      if (!db.objectStoreNames.contains('settings')) db.createObjectStore('settings', { keyPath: 'key' });

    };

    request.onsuccess = (e) => { 

      db = e.target.result; 

      loadRemainingLives();

      checkMidnightEvaluation();

    };

    request.onerror = (e) => console.error("IndexedDB error:", e);

  }


  let remainingLives = 3;

  function loadRemainingLives() {

    getStoredSetting('remainingLives', (val) => {

      if (val !== null) remainingLives = parseInt(val, 10);

      else saveRemainingLives(3);

    });

  }

  function saveRemainingLives(count) {

    remainingLives = count;

    const transaction = db.transaction(['settings'], 'readwrite');

    transaction.objectStore('settings').put({ key: 'remainingLives', value: String(count) });

  }


  function checkMidnightEvaluation() {

    const todayStr = new Date().toISOString().split('T')[0];

    getStoredSetting('lastCheckedDate', (lastDate) => {

      if (!lastDate) {

        saveSettingDirect('lastCheckedDate', todayStr);

        return;

      }

      if (lastDate !== todayStr) {

        evaluateDayResult(lastDate, () => {

          saveSettingDirect('lastCheckedDate', todayStr);

        });

      }

    });

  }


  function evaluateDayResult(targetDateStr, callback) {

    const transaction = db.transaction(['tasks'], 'readonly');

    transaction.objectStore('tasks').getAll().onsuccess = (e) => {

      const allTasks = e.target.result;

      const dayTasks = allTasks.filter(t => t.date === targetDateStr);


      let title = "";

      let art = "";

      let desc = "";


      if (dayTasks.length > 0) {

        const allCompleted = dayTasks.every(t => t.completed);

        if (allCompleted) {

          title = "항해 성공";

          art = "     |    \n   --+--  \n     |    \n ~ ^ ~ ^ ~";

          desc = `오디세우스는 ${targetDateStr} 유혹을 무사히 견뎌냈습니다.`;

        } else {

          title = "항해 실패";

          art = "    \\ o / \n     \\|/  \n     / \\  \n ~ ~ ~ ~ ~";

          desc = `오디세우스는 ${targetDateStr} 유혹을 참지 못하고 세이렌을 향해 헤엄쳤습니다.`;

          

          remainingLives = Math.max(0, remainingLives - 1);

          saveRemainingLives(remainingLives);


          if (remainingLives === 0) {

            triggerPenaltyEmail();

          }

        }

      } else {

        if (callback) callback();

        return;

      }


      showResultPopup(title, art, desc);

      if (callback) callback();

    };

  }


  function showResultPopup(title, art, desc) {

    document.getElementById('resultPopupTitle').innerText = title;

    document.getElementById('resultPopupArt').innerText = art;

    document.getElementById('resultPopupDesc').innerText = desc;

    document.getElementById('resultPopup').classList.remove('hidden');

  }


  function closeResultPopup() {

    document.getElementById('resultPopup').classList.add('hidden');

  }


  function saveSettingDirect(key, value) {

    if (!db) return;

    const transaction = db.transaction(['settings'], 'readwrite');

    transaction.objectStore('settings').put({ key: key, value: value });

  }

  function toggleEditMode() {

    if (isUnlocked) {

      isUnlocked = false;

      updateEditLockButtonUI();

      switchView(currentView);

    } else {

        if (!savedPw) {

            isUnlocked = true;

            updateEditLockButtonUI();

            switchView(currentView);

          }

          return;

        }

        const inputPw = prompt("수정 권한을 얻으려면 비밀번호를 입력하세요:");

        if (inputPw === savedPw) {

          isUnlocked = true;

          updateEditLockButtonUI();

          alert("수정 권한이 인증되었습니다.");

          switchView(currentView);

        } else if (inputPw !== null) {

          alert("비밀번호가 일치하지 않습니다.");

        }

      });

    }

  }


  function updateEditLockButtonUI() {

    const btn = document.getElementById('editLockBtn');

    if (isUnlocked) {

      btn.innerText = " 수정 완료(잠금)";

      btn.classList.add('unlocked');

    } else {

      btn.innerText = " 잠김";

      btn.classList.remove('unlocked');

    }

  }


  function getStoredSetting(key, callback) {

    if (!db) { callback(null); return; }

    const transaction = db.transaction(['settings'], 'readonly');

    const store = transaction.objectStore('settings');

    const req = store.get(key);

    req.onsuccess = (e) => {

      const result = e.target.result;

      callback(result ? result.value : null);

    };

    req.onerror = () => callback(null);

  }


  function getMonday(d) {

    d = new Date(d);

    let day = d.getDay(), diff = d.getDate() - day + (day === 0 ? -6 : 1);

    return new Date(d.setDate(diff));

  }


  function formatDateStr(d) {

    return d.toISOString().split('T')[0];

  }


  function startRunningRecovery() {

    getStoredSetting('geminiKey', (apiKey) => {

      if (!apiKey) {

        alert("Gemini API Key가 설정되어 있지 않습니다.");

        switchView('settings');

        return;

      }

      if (remainingLives >= 3) {

        alert("기회가 가득 차 있습니다. (3/3)");

        return;

      }

      isRunningRecoveryMode = true;

      document.getElementById('aiImageInput').click();

    });

  }


  function triggerPenaltyEmail() {

    getStoredSetting('emailService', (serviceId) => {

      getStoredSetting('emailTemplate', (templateId) => {

        getStoredSetting('emailPublic', (publicKey) => {

          getStoredSetting('targetEmail', (targetEmail) => {

            if (!serviceId || !templateId || !publicKey || !targetEmail) return;

            emailjs.init(publicKey);

            emailjs.send(serviceId, templateId, {

              to_email: targetEmail,

              message: "오디세우스 프로젝트 실패 횟수 초과로 페널티(커피)가 발생했습니다!"

            });

          });

        });

      });

    });

  }


  function handleImageUpload(event) {

    const file = event.target.files[0];

    if (!file) return;


    if (isRunningRecoveryMode) {

      isRunningRecoveryMode = false;

      event.target.value = '';

      getStoredSetting('geminiKey', async (apiKey) => {

        document.getElementById('loadingText').innerText = "러닝 기록 판별 중...";

        document.getElementById('loadingOverlay').classList.remove('hidden');

        try {

          const base64Data = await fileToBase64(file);

          const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`, {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ contents: [{ parts: [{ text: "10km 러닝 완주 증명 사진인지 판단. 첫 줄에 PASS 또는 FAIL 출력." }, { inline_data: { mime_type: file.type, data: base64Data } }] }] })

          });

          const data = await response.json();

          document.getElementById('loadingOverlay').classList.add('hidden');

          if (data.candidates && data.candidates[0].content.parts[0].text.toUpperCase().includes('PASS')) {

            saveRemainingLives(Math.min(3, remainingLives + 1));

            alert("러닝 복구 성공!");

            if (currentView === 'daily') renderDailyView(document.getElementById('appContent'));

          } else {

            alert("러닝 인증 실패");

          }

        } catch (err) {

          document.getElementById('loadingOverlay').classList.add('hidden');

          alert("오류 발생");

        }

      });

      return;

    }


    if (!activeTaskIdForAI) return;

    const taskId = activeTaskIdForAI;

    activeTaskIdForAI = null;

    event.target.value = '';


    db.transaction(['tasks'], 'readonly').objectStore('tasks').get(taskId).onsuccess = (e) => {

      const task = e.target.result;

      if (!task) return;

      getStoredSetting('geminiKey', async (apiKey) => {

        document.getElementById('loadingText').innerText = "이미지 판별 중...";

        document.getElementById('loadingOverlay').classList.remove('hidden');

        try {

          const base64Data = await fileToBase64(file);

          const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`, {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ contents: [{ parts: [{ text: `[제목: ${task.title}] [기준: ${task.criteria}] 부합 여부 판별. 첫 줄에 PASS 또는 FAIL 출력.` }, { inline_data: { mime_type: file.type, data: base64Data } }] }] })

          });

          const data = await response.json();

          document.getElementById('loadingOverlay').classList.add('hidden');

          if (data.candidates && data.candidates[0].content.parts[0].text.toUpperCase().includes('PASS')) {

            alert("인증 성공");

            updateTaskStatus(taskId, true);

          } else {

            alert("인증 실패");

          }

        } catch (err) {

          document.getElementById('loadingOverlay').classList.add('hidden');

          alert("오류 발생");

        }

      });

    };

  }


  function fileToBase64(file) {

    return new Promise((resolve, reject) => {

      const reader = new FileReader();

      reader.readAsDataURL(file);

      reader.onload = () => resolve(reader.result.split(',')[1]);

      reader.onerror = error => reject(error);

    });

  }


  function handleTaskVerification(task) {

    if (task.completed) {

      if (confirm("해제하시겠습니까?")) updateTaskStatus(task.id, false);

      return;

    }

    if (task.type === 'simple') updateTaskStatus(task.id, true);

    else if (task.type === 'location') verifyLocation(task);

    else if (task.type === 'ai') verifyAIImage(task);

  }


  function verifyLocation(task) {

    if (!navigator.geolocation) { alert("위치 미지원"); return; }

    navigator.geolocation.getCurrentPosition(

      (pos) => { alert("위치 인증 성공"); updateTaskStatus(task.id, true); },

      (err) => { alert("위치 실패: " + err.message); }

    );

  }


  function verifyAIImage(task) {

    getStoredSetting('geminiKey', (apiKey) => {

      if (!apiKey) { alert("API Key 필요"); switchView('settings'); return; }

      activeTaskIdForAI = task.id;

      document.getElementById('aiImageInput').click();

    });

  }


  function updateTaskStatus(id, status) {

    const transaction = db.transaction(['tasks'], 'readwrite');

    const store = transaction.objectStore('tasks');

    store.get(id).onsuccess = (e) => {

      let task = e.target.result;

      if (task) {

        task.completed = status;

        store.put(task);

        if (currentView === 'daily') loadTasksForCurrentDate();

        else if (currentView === 'weekly') loadTasksForWeeklyView(currentWeekStart);

        else if (currentView === 'monthly') renderMonthlyView(document.getElementById('appContent'));

      }

    };

  }


  function renderDailyView(container) {

      <div class="task-form">

        <input type="text" id="taskTitle" placeholder="계획 제목">

        <select id="taskType" onchange="toggleCriteriaInput(this.value)">

          <option value="simple">단순 체크</option>

          <option value="location">위치 인증</option>

          <option value="ai">AI 이미지 인증</option>

        </select>

        <input type="text" id="taskCriteria" class="hidden" placeholder="인증 기준 입력">

        <button onclick="addNewTask()" style="width: 100%; background: #28a745; color: white; border: none; padding: 10px; border-radius: 4px;">추가하기</button>

      </div>

    ` : `<div class="locked-notice"> 잠금 상태입니다.</div>`;


    container.innerHTML = `

      <div class="life-status-box">

        <div> 남은 기회: <strong>${remainingLives} / 3</strong></div>

        <button onclick="startRunningRecovery()" style="background:#17a2b8; color:white; border:none; padding:6px 12px; border-radius:4px; font-weight:bold;">♂ 10km 복구</button>

      </div>

      <div class="header-controls">

        <button onclick="changeDailyDate(-1)">&lt; 이전</button>

        <input type="date" id="selectedDateInput" value="${currentDate}" onchange="onDateChange(this.value)">

        <button onclick="changeDailyDate(1)">다음 &gt;</button>

      </div>

      ${formHtml}

      <div id="taskListContainer"></div>

    `;

  }


  function toggleCriteriaInput(type) {

    const criteriaInput = document.getElementById('taskCriteria');

    if (!criteriaInput) return;

    if (type === 'simple') criteriaInput.classList.add('hidden');

    else criteriaInput.classList.remove('hidden');

  }


  function changeDailyDate(days) {

    let d = new Date(currentDate);

    d.setDate(d.getDate() + days);

    currentDate = formatDateStr(d);

    const input = document.getElementById('selectedDateInput');

    if (input) input.value = currentDate;

    loadTasksForCurrentDate();

  }


  function onDateChange(val) { currentDate = val; loadTasksForCurrentDate(); }


  function loadTasksForCurrentDate() {

    if (!db) return;

    db.transaction(['tasks'], 'readonly').objectStore('tasks').getAll().onsuccess = (e) => {

      const allTasks = e.target.result;

      const filtered = allTasks.filter(t => t.date === currentDate).sort((a, b) => (a.order || 0) - (b.order || 0));

      const listContainer = document.getElementById('taskListContainer');

      if (!listContainer) return;


      if (filtered.length === 0) {

        listContainer.innerHTML = `<p style="text-align: center; color: #777;">계획 없음</p>`;

        return;

      }


      let html = `<ul class="task-list">`;

      filtered.forEach((task, index) => {

        let actionButtons = isUnlocked ? `

          <div style="display:flex; gap:4px;">

            <button onclick="moveTask(${task.id}, 'up')" ${index === 0 ? 'disabled' : ''}>▲</button>

            <button onclick="moveTask(${task.id}, 'down')" ${index === filtered.length - 1 ? 'disabled' : ''}>▼</button>

            <button onclick="deleteTask(${task.id})" style="background: #dc3545; color: white; border: none;">삭제</button>

          </div>

        ` : '';

          <li class="task-item">

'#6c757d' : '#007bff'}; color: white; border: none; padding: 6px 12px; border-radius: 4px; font-weight: bold;">${task.completed ? ' 완료' : '미완료'}</button>

          </li>

        `;

      });

  }


  function addNewTask() {

    if (!isUnlocked) return;

    const title = document.getElementById('taskTitle').value.trim();

    const type = document.getElementById('taskType').value;

    const criteria = document.getElementById('taskCriteria').value.trim();

    if (!title) return;


    const transaction = db.transaction(['tasks'], 'readwrite');

    const store = transaction.objectStore('tasks');

    store.getAll().onsuccess = (e) => {

      const tasks = e.target.result.filter(t => t.date === currentDate);

      const maxOrder = tasks.length > 0 ? Math.max(...tasks.map(t => t.order || 0)) : -1;

      store.add({ title, type, criteria, date: currentDate, completed: false, order: maxOrder + 1 })

        .onsuccess = () => loadTasksForCurrentDate();

    };

  }


  function deleteTask(id) {

    if (!isUnlocked) return;

    if (!confirm("삭제하시겠습니까?")) return;

    db.transaction(['tasks'], 'readwrite').objectStore('tasks').delete(id).onsuccess = () => loadTasksForCurrentDate();

  }


  function moveTask(id, direction) {

    if (!isUnlocked) return;

    const transaction = db.transaction(['tasks'], 'readwrite');

    const store = transaction.objectStore('tasks');

    store.getAll().onsuccess = (e) => {

      const tasks = e.target.result.filter(t => t.date === currentDate).sort((a, b) => (a.order || 0) - (b.order || 0));

      const idx = tasks.findIndex(t => t.id === id);

      if (idx === -1) return;

      const targetIdx = direction === 'up' ? idx - 1 : idx + 1;

      if (targetIdx < 0 || targetIdx >= tasks.length) return;


      const tempOrder = tasks[idx].order;

      tasks[idx].order = tasks[targetIdx].order;

      tasks[targetIdx].order = tempOrder;

      store.put(tasks[idx]);

      store.put(tasks[targetIdx]);

      transaction.oncomplete = () => loadTasksForCurrentDate();

    };

  }

  function loadTasksForWeeklyView(startDate) {

    if (!db) return;

    let weekDates = [];

    for (let i = 0; i < 7; i++) {

      let d = new Date(startDate);

      d.setDate(d.getDate() + i);

      weekDates.push(formatDateStr(d));

    }

    db.transaction(['tasks'], 'readonly').objectStore('tasks').getAll().onsuccess = (e) => {

      const allTasks = e.target.result;

      const grid = document.getElementById('weeklyGrid');

      if (!grid) return;

      let gridHtml = '';

      const dayNames = ['월', '화', '수', '목', '금', '토', '일'];

      weekDates.forEach((dateStr, index) => {

        const dayTasks = allTasks.filter(t => t.date === dateStr).sort((a, b) => (a.order || 0) - (b.order || 0));

        gridHtml += `

          <div class="week-day-card">

            <div class="week-day-header">${dayNames[index]}<br><span style="font-size:0.75em; color:#666;">${dateStr.slice(5)}</span></div>

            <div class="week-day-tasks">

        `;

        if (dayTasks.length === 0) gridHtml += `<div style="color:#aaa; text-align:center; margin-top:20px; font-size:0.8em;">없음</div>`;

        else {

          dayTasks.forEach(task => {

            gridHtml += `<div class="week-task-item ${task.completed ? 'done' : ''}" onclick="jumpToDate('${dateStr}')"><span>${task.completed ? '' : ''}</span><span style="overflow:hidden; text-overflow:ellipsis; white-space:nowrap;">${escapeHtml(task.title)}</span></div>`;

          });

        }

        gridHtml += `</div></div>`;

      });

      grid.innerHTML = gridHtml;

    };

  }

  function loadTasksForMonthlyView(year, month) {

    if (!db) return;

    const firstDayIndex = (new Date(year, month, 1).getDay() + 6) % 7;

    const lastDate = new Date(year, month + 1, 0).getDate();

    db.transaction(['tasks'], 'readonly').objectStore('tasks').getAll().onsuccess = (e) => {

      const allTasks = e.target.result;

      const grid = document.getElementById('calendarGrid');

      if (!grid) return;

      let gridHtml = `<div class="calendar-weekday">월</div><div class="calendar-weekday">화</div><div class="calendar-weekday">수</div><div class="calendar-weekday">목</div><div class="calendar-weekday">금</div><div class="calendar-weekday">토</div><div class="calendar-weekday">일</div>`;

      for (let i = 0; i < firstDayIndex; i++) gridHtml += `<div class="calendar-cell empty"></div>`;

      for (let day = 1; day <= lastDate; day++) {

        const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;

        const dayTasks = allTasks.filter(t => t.date === dateStr).sort((a, b) => (a.order || 0) - (b.order || 0));

        gridHtml += `<div class="calendar-cell" onclick="jumpToDate('${dateStr}')"><div class="calendar-date-num">${day}</div><div class="calendar-tasks-preview">`;

        dayTasks.forEach(task => { gridHtml += `<div class="cal-task-badge ${task.completed ? 'done' : ''}">${escapeHtml(task.title)}</div>`; });

        gridHtml += `</div></div>`;

      }

      grid.innerHTML = gridHtml;

    };

  }

  function saveSettings() {

    const pass = document.getElementById('setPass').value.trim();

    const geminiKey = document.getElementById('setGeminiKey').value.trim();

    const emailService = document.getElementById('setEmailService').value.trim();

    const emailTemplate = document.getElementById('setEmailTemplate').value.trim();

    const emailPublic = document.getElementById('setEmailPublic').value.trim();

    const targetEmail = document.getElementById('setTargetEmail').value.trim();


    const transaction = db.transaction(['settings'], 'readwrite');

    if (pass) store.put({ key: 'password', value: pass });

  }
